#include<stdio.h>
#include<string.h>
int main(){
    int x=10;
    int *ptr;
    ptr=&x;
    printf("Address of x: %p",ptr);
    printf("\nvalue of x by pointer reference: %d", *ptr);
    x=15;
    *ptr=25;
    printf("\nvalue of x by pointer reference: %d", x);
    return 0;
}

#include<stdio.h>
#include<string.h>
int main(){
    int arr[]={10,20,30,40,50};   //Accessing array using ptr
    printf(" %d", *arr);      //10
    printf("\n %d", *(arr+1));//20           //Pointer Arithmatic
    printf("\n %d", *(arr+2));//30
    printf("\n %d", *arr+3);  //13
    return 0;
}
