#include<stdio.h>
#include<string.h>
int main(){
    int arr[]={6,3,1,8,9,2}; //{1,2,3,6,8,9} 
    int n=sizeof(arr)/sizeof(arr[0]);
    int a;
    for (int i=0; i< n-1;i++){
        for (int j=0; j < n-1;j++){            //Bubble sorting
            if (arr[j]>arr[j+1]){
                a=arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=a;
            }
        }
    }
    for (int x=0;x<n;x++){
        printf("%d ", arr[x]);
    }
    return 0;
}

int main(){
    int arr[]={6,3,1,8,9,2}; //{1,2,3,6,8,9}           //combination of selection sorting and bubble sorting- less iterations
    int n=sizeof(arr)/sizeof(arr[0]);
    int a;
    for (int i=0; i< n-1;i++){
        for (int j=0; j < n-i-1;j++){
            if (arr[j]>arr[j+1]){
                a=arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=a;
            }
        }
    }
    for (int x=0;x<n;x++){
        printf("%d ", arr[x]);
    }
    return 0;
}
