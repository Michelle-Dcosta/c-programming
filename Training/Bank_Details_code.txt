#include<stdio.h>
#include<stdlib.h>
#include<string.h>
struct account_details{
    char account_number[10];
    float balance;
    int pin;
};
int main(){
    int n,i;
    printf("Number of Accounts ");
    scanf("%d",&n);
    struct account_details c[n];
    FILE *fp;
    fp=fopen("Bank Details.txt","w");
    for(i=0;i<n;i++){
    printf("Account Number:");
    scanf("%s",&c[i].account_number);
    printf("Balance:");
    scanf("%f",&c[i].balance);
    printf("Enter the pin:");
    scanf("%d",&c[i].pin);
    fprintf(fp,"%s","Account Number:");
    fprintf(fp,"%s\n",c[i].account_number);
    fprintf(fp,"%s","Balance::");
    fprintf(fp,"%f\n", c[i].balance);
    fprintf(fp,"%s","pin:");
    fprintf(fp,"%d\n", c[i].pin);
}
    fclose(fp);
    return 0;
}
