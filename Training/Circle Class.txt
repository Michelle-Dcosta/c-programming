#include<iostream>
using namespace std;
class circle{
    public:
    float pi=3.14;
    int radius;
    circle(int r){
        radius=r;
    }
    float area(){
        float a=pi*radius*radius;
        return a;
    }
    float peri(){
        float p=2*pi*radius;
        return p;
    }
    ~circle(){
        cout<<"Destucted"<<endl;
    }
};
int main(){
    circle c(5);
    cout<<"The area is: "<<c.area()<<endl;
    cout<<"The perimeter is: "<<c.peri()<<endl;
    circle c1(7);
    cout<<"The area is: "<<c1.area()<<endl;
    cout<<"The perimeter is: "<<c1.peri()<<endl;
    circle c2(10);
    cout<<"The area is: "<<c2.area()<<endl;
    cout<<"The perimeter is: "<<c2.peri()<<endl;
    cout<<"Main Ended"<<endl;
}
