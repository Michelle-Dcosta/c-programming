#include<stdio.h>
#include<string.h>
int main(){
    char n[]="4567897";
    int nch;
    int total=0;
    short int len=strlen(n);
    for(int i=0; i<len; i++){
        nch=n[i]-48;
        total+=nch;
    }
    printf("sum of digit:%d",total);
    return 0;
}
