#include<stdio.h>
#include<string.h>
int main(){
    char *countries[3]={"India","USA","Argentina"};
    int n=sizeof(countries)/sizeof(countries[0]);
    char *arr[1];
    for (int i=0; i< n;i++){
        for (int j=0; j < n-1;j++){            //Bubble sorting
            if (countries[j][0]>countries[j+1][0]){
                arr[0] = countries[j];
                countries[j] = countries[j+1];
                countries[j+1] =arr[0];
            }
        }
    }
    for (int x=0;x<n;x++){
        printf(" %s ", countries[x]);
    }
    return 0;
}
