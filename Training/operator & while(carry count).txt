#include <stdio.h>
int main() {
//1 1 1 ---------3
//8 7 8 6
//  7 8 5(+)
//9 5 7 1
int n1=8786;
int n2=785;
int l1,l2,s;
int count=0;
int carry=0;
while(n1>0 || n2>0){
   l1=n1%10;
   l2=n2%10;
   s=l1+l2+carry;
    if (s>9){
        count++;
        carry=1;
    }else{carry=0;}
    n1=n1/10;
    n2=n2/10;
}
printf("count: %d",count);
return 0;
}
count: 3
