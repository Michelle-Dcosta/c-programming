#include<stdio.h>
int main(){
    int arr[]={7,4,9,1,3,2};
    int len=sizeof(arr)/sizeof(arr[0]);
    int minvalue=arr[0];
    int profit,maxprofit=0;
    for (int i=0;i<len;i++){
        if (arr[i] < minvalue){
            minvalue = arr[i];
        }
        profit = arr[i]-minvalue;
        if(profit>maxprofit){
            maxprofit=profit;
        }}
        printf("Max profit: %d",maxprofit);
    return 0;
}
