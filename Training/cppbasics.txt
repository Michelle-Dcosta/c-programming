#include<iostream>
using namespace std;
class student{
    public: //Access specifier
    string name; //Global
    int age;
    void display(){
        cout<<"The name is "<<name<<endl;
        cout<<"The age is "<<age<<endl;
    }
};
int main(){
    student s1;
    s1.name="Michelle";
    s1.age=21;
    s1.display();
}
The name is Michelle
The age is 21
