#include<stdio.h>
int main(){
    int arr[5]={10,20,30,40,50};
    short int size_of_arr=sizeof(arr);
    printf("size of arr: %d",size_of_arr);
    int arr1[5]={10,20,30};
    short int size_of_arr1=sizeof(arr1);
    printf("\nsize of arr1: %d",size_of_arr1);
    int arr2[5];
    short int size_of_arr2=sizeof(arr2);
    printf("\nsize of arr2: %d",size_of_arr2);
    int arr3[]={10,20,30,40,50,60};
    short int size_of_arr3=sizeof(arr3);
    printf("\nsize of arr3: %d",size_of_arr3);
    //length of array
    int len=sizeof(arr3)/sizeof(arr3[0]);
    printf("\nlen of arr3: %d",len);
    //Accessing value
    int arr4[5]={10,20,30,40,50};
    printf("\n5th index of arr4: %d",arr4[5]);
    int arr5[5]={10,20};
    printf("\n2nd index of arr5: %d",arr5[2]);
    return 0;
}
