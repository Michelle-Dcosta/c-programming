#include<stdio.h>
#include<stdlib.h>
#include<string.h>
struct Book{
    char title[100];
    char author[100];
    float price;
};
int main(){
    printf("enter the number of entries:");
    int n;
    scanf("%d",&n);
    int i;
    struct Book b[n];
    for(i=0;i<n;i++){
        printf("Title:");
        scanf("%s",b[i].title);
        printf("Author Name:");
        scanf("%s",b[i].author);
        printf("Price:");
        scanf("%d",&b[i].price);
    }
    for(i=0;i<n;i++){
        printf("Title: %s\n",b[i].title);
        printf("Author Name: %s\n",b[i].author);
        printf("Price: %d\n",b[i].price);
    }
    return 0;
}
