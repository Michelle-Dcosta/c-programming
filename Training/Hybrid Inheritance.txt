#include<iostream>
using namespace std;               //Diamond problem
class b{
    public:
    string name;
};
class D1: virtual public b{
    public :
    int age;
    void display(){
        cout<<"The name is: "<<name<<" age is: "<<age<<endl;
    }
};
class D2: virtual public b{
    public :
    int sal;
    void display1(){
        cout<<"The name is: "<<name<<" salary is: "<<sal<<endl;
    }
};
class sder: public D1,public D2{
    public:
    int ID;
    void display2(){
        cout<<"The Employee name is: "<<name<<" age is: "<<age<<" ID is: "<<ID<<" salary is: "<<sal<<endl;
    }
};
int main(){
    sder s1;
    s1.name="Ram";
    s1.age=21;
    s1.sal=70000;
    s1.ID=903;
    s1.display();
    s1.display1();
    s1.display2();
}
The name is: Ram age is: 21
The name is: Ram salary is: 70000
The Employee name is: Ram age is: 21 ID is: 903 salary is: 70000
